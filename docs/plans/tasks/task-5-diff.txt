+//! SQLite store: schema creation, index-meta validation, and CRUD.
+//!
+//! sqlite-vec is registered in-process before any connection opens (no
+//! runtime `.so` load); registration failure is fatal. Opening an index
+//! whose `model_id`, `dimensions`, or `prefix_profile` differ from the
+//! merged config is fatal — the user must pass a new `--index` or delete the
+//! index file. A `backend`-only difference is a stderr warning, not an
+//! error. There is deliberately no method that rewrites meta (`ferret rm`
+//! must not change it).

+/// Cache of the `sqlite3_auto_extension` return code so the C extension is
+/// registered exactly once per process, before the first connection opens.
+static SQLITE_VEC_REGISTRATION: OnceLock<i32> = OnceLock::new();
+
+fn register_sqlite_vec() -> Result<()> {
+    let rc = *SQLITE_VEC_REGISTRATION.get_or_init(|| unsafe {
+        rusqlite::ffi::sqlite3_auto_extension(Some(std::mem::transmute::<
+            *const (),
+            unsafe extern "C" fn(
+                *mut rusqlite::ffi::sqlite3,
+                *mut *mut std::os::raw::c_char,
+                *const rusqlite::ffi::sqlite3_api_routines,
+            ) -> std::os::raw::c_int,
+        >(
+            sqlite_vec::sqlite3_vec_init as *const (),
+        )))
+    });
+    if rc != rusqlite::ffi::SQLITE_OK {
+        return Err(Error::fatal(format!(
+            "failed to register sqlite-vec in-process (sqlite3_auto_extension rc = {rc}); \
+             the sqlite-vec C extension did not build or link"
+        )));
+    }
+    Ok(())
+}
+
+/// Open (creating if absent) the index at `path` and validate its meta
+/// against the merged config.
+pub fn open(path: &Path, cfg: &Config, images_supported: bool) -> Result<Store> {
+    register_sqlite_vec()?;
+    let new_index = !path.exists();
+    let conn = Connection::open(path)?;
+    let store = Store {
+        conn,
+        dimensions: cfg.embedder.dimensions,
+    };
+    store.apply_pragmas()?;
+    store.register_functions()?;
+    if new_index {
+        store.conn.execute_batch(&schema::ddl(cfg.embedder.dimensions))?;
+        store.write_initial_meta(cfg, images_supported)?;
+    } else {
+        store.validate_meta(cfg, path)?;
+    }
+    Ok(store)
+}

+impl Store {
+    /// Read-only connection handle for tests and read-only CLI output; all
+    /// writes go through the methods above.
+    pub fn conn(&self) -> &Connection {
+        &self.conn
+    }
+
+    /// WAL, a 5s busy timeout, and enforced foreign keys; also refuses
+    /// SQLite older than 3.38.
+    fn apply_pragmas(&self) -> Result<()> {
+        let version: String =
+            self.conn.query_row("select sqlite_version()", [], |r| r.get(0))?;
+        if !sqlite_at_least(&version, 3, 38) {
+            return Err(Error::fatal(format!(
+                "SQLite {version} is too old; ferret requires >= 3.38.0 (rusqlite bundled)"
+            )));
+        }
+        let mode: String = self
+            .conn
+            .pragma_update_and_check(None, "journal_mode", "WAL", |r| r.get(0))?;

+                "could not enable WAL journal mode (got `{mode}`)"
+            )));
+        }
+        self.conn.busy_timeout(Duration::from_millis(5000))?;
+        self.conn.pragma_update(None, "foreign_keys", "ON")?;
+        Ok(())
+    }

+    fn validate_meta(&self, cfg: &Config, path: &Path) -> Result<()> {
+        let has_meta: bool = self.conn.query_row(
+            "SELECT EXISTS(SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'meta')",
+            [],
+            |r| r.get(0),
+        )?;
+        if !has_meta {
+            return Err(Error::fatal(format!(
+                "{} is not a ferret index (no meta table); pass a new --index path or delete the file",
+                path.display()
+            )));
+        }
+
+        let expected = [
+            ("model_id", cfg.embedder.model.clone()),
+            ("dimensions", cfg.embedder.dimensions.to_string()),
+            ("prefix_profile", cfg.embedder.prefix_profile.to_string()),
+        ];
+        for (key, want) in expected {
+            let found = self.get_meta(key)?;

+                backend.as_deref().unwrap_or("<missing>"),

+        }
+        Ok(())

+    pub fn delete_file_by_path(&self, path: &str) -> Result<bool> {

+        // foreign keys, so clean them before the files delete cascades.
+        for table in ["chunks_vec", "chunks_fts"] {
+            self.conn.execute(
+                &format!(
+                    "DELETE FROM {table} WHERE rowid IN \
+                     (SELECT c.id FROM chunks c JOIN files f ON f.id = c.file_id \

+                [path],

+        }
+        let deleted = self
+            .conn
+            .execute("DELETE FROM files WHERE path = ?1", [path])?;
+        Ok(deleted > 0)

+    pub fn insert_chunk(&self, tx: &Transaction<'_>, chunk: NewChunk) -> Result<i64> {
+        // Empty text is stored as SQL NULL.
+        let text = chunk.text.as_deref().filter(|t| !t.is_empty());
+        tx.execute(
+            "INSERT INTO chunks (file_id, ordinal, page, kind, text, start_byte, end_byte)
+             VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7)",
+            params![
+                chunk.file_id,
+                chunk.ordinal,
+                chunk.page,
+                chunk.kind.to_string(),
+                text,
+                chunk.start_byte,
+                chunk.end_byte,
+            ],
+        )?;
+        Ok(tx.last_insert_rowid())
+    }
+
+    /// Insert one embedding; `rowid` is the `chunks.id` and `file_kind` is
+    /// the FILES kind (text|image|pdf), never the chunk kind.
+    #[allow(clippy::too_many_arguments)] // signature fixed by the task brief
+    pub fn insert_vec(
+        &self,
+        tx: &Transaction<'_>,
+        chunk_id: i64,
+        embedding: &[f32],
+        file_kind: FileKind,
+        ext: &str,
+        mtime: i64,
+        collection: &str,
+    ) -> Result<()> {
+        // Global constraint: store only finite f32.
+        if let Some(pos) = embedding.iter().position(|f| !f.is_finite()) {
+            return Err(Error::fatal(format!(
+                "refusing to store non-finite embedding value at index {pos} \
+                 for chunk {chunk_id}"
+            )));
+        }
+        if embedding.len() as u32 != self.dimensions {
+            return Err(Error::fatal(format!(
+                "embedding length {} does not match index dimensions {}",
+                embedding.len(),
+                self.dimensions
+            )));
+        }
+        tx.execute(
+            "INSERT INTO chunks_vec (rowid, embedding, file_kind, ext, mtime, collection)
+             VALUES (?1, ?2, ?3, ?4, ?5, ?6)",
+            params![
+                chunk_id,
+                f32_blob(embedding),
+                file_kind.to_string(),
+                ext,
+                mtime,
+                collection,
+            ],
+        )?;
+        Ok(())
+    }

+    /// Delete a file's `text` chunks (keeping `image` and `pdf_page`).
+    pub fn delete_text_chunks_keep_images(&self, file_id: i64) -> Result<()> {
+        self.delete_chunks_where(file_id, "kind = 'text'")
+    }

+    /// Delete a file's `image` and `pdf_page` chunks (keeping `text`).
+    pub fn delete_image_chunks_keep_text(&self, file_id: i64) -> Result<()> {
+        self.delete_chunks_where(file_id, "kind IN ('image', 'pdf_page')")

+    /// `kind_filter` is always a compile-time literal, never user input.
+    fn delete_chunks_where(&self, file_id: i64, kind_filter: &str) -> Result<()> {
+        for table in ["chunks_vec", "chunks_fts"] {
+            self.conn.execute(
+                &format!(
+                    "DELETE FROM {table} WHERE rowid IN \
+                     (SELECT id FROM chunks WHERE file_id = ?1 AND {kind_filter})"
+                ),
+                [file_id],
+            )?;
+        }
+        self.conn.execute(
+            &format!("DELETE FROM chunks WHERE file_id = ?1 AND {kind_filter}"),
+            [file_id],
+        )?;
+        Ok(())
+    }

+    pub fn status_counts(&self) -> Result<StatusCounts> {
+        let count = |sql: &str| -> Result<u64> {
+            let n: i64 = self.conn.query_row(sql, [], |r| r.get(0))?;
+            Ok(n as u64)
+        };
+        let indexed_at_max: Option<i64> =
+            self.conn.query_row("SELECT max(indexed_at) FROM files", [], |r| r.get(0))?;
+        Ok(StatusCounts {
+            files_text: count("SELECT count(*) FROM files WHERE kind = 'text'")?,
+            files_image: count("SELECT count(*) FROM files WHERE kind = 'image'")?,
+            files_pdf: count("SELECT count(*) FROM files WHERE kind = 'pdf'")?,
+            chunks: count("SELECT count(*) FROM chunks")?,
+            incomplete: count("SELECT count(*) FROM files WHERE complete = 0")?,
+            errors: count("SELECT count(*) FROM files WHERE error IS NOT NULL")?,
+            indexed_at_max,
+        })
+    }
+
+    /// `&self`, not `&mut self`: callers keep using the store's `&self`
+    /// methods (`upsert_file`, `insert_chunk`, `insert_vec`) while the
+    /// transaction is open, so an exclusive borrow here would not compile
+    /// at those call sites. `new_unchecked` trades the compile-time
+    /// no-nesting guarantee for a runtime one (BEGIN fails if a transaction
+    /// is already open on this connection).
+    pub fn begin(&self) -> Result<Transaction<'_>> {
+        Ok(Transaction::new_unchecked(
+            &self.conn,
+            rusqlite::TransactionBehavior::Deferred,
+        )?)
+    }

+/// `"3.38.0"`-style version at or above `major.minor`.
+fn sqlite_at_least(version: &str, want_major: u32, want_minor: u32) -> bool {
+    let mut parts = version.split('.');
+    let major: u32 = parts.next().unwrap_or("0").parse().unwrap_or(0);
+    let minor: u32 = parts.next().unwrap_or("0").parse().unwrap_or(0);
+    (major, minor) >= (want_major, want_minor)
+}

+/// The hard-error message for a model/dimensions/prefix_profile mismatch.
+fn meta_mismatch(path: &Path, key: &str, found: Option<&str>, expected: &str) -> Error {
+    Error::fatal(format!(
+        "index {} mismatch: this index was built with `{}` but the merged config has \
+         `{expected}`; pass a new --index path or delete the index file {} to rebuild",
+        key,
+        found.unwrap_or("<missing>"),
+        path.display()
+    ))
+}

+/// sqlite-vec reads `FLOAT[d]` blobs as a C `float` array, so the byte order
+/// is the platform's native order.
+fn f32_blob(values: &[f32]) -> Vec<u8> {
+    let mut bytes = Vec::with_capacity(values.len() * 4);
+    for value in values {
+        bytes.extend_from_slice(&value.to_ne_bytes());
+    }
+    bytes

+//! Index schema DDL (spec §5.1).
+//!
+//! `dimensions` is interpolated only into the vec0 embedding width
+//! (`FLOAT[{d}]`); no path or other user input is ever concatenated into
+//! this SQL.
+//!
+//! The `chunks_fts` triggers (after insert/update/delete on `chunks`, and a
+//! rewrite when `files.path` changes) land with indexing, not here; they use
+//! the `ferret_basename(path)` scalar function that `store::open` registers
+//! on every connection.
+
+/// Complete DDL for a new index: `meta`, `files`, `chunks`, the FTS5 index
+/// over chunk text + filename + path, and the `vec0` vector table whose
+/// `rowid` mirrors `chunks.id`. Its `file_kind` metadata column is the
+/// `files.kind` (text|image|pdf), never the chunk kind.
+pub fn ddl(dimensions: u32) -> String {
+    let d = dimensions;
+    format!(
+        r#"
+CREATE TABLE meta (key TEXT PRIMARY KEY, value TEXT NOT NULL);
+CREATE TABLE files (
+  id INTEGER PRIMARY KEY,
+  path TEXT NOT NULL UNIQUE,
+  collection TEXT NOT NULL,
+  kind TEXT NOT NULL,
+  ext TEXT NOT NULL,
+  size INTEGER NOT NULL,
+  mtime INTEGER NOT NULL,
+  fingerprint TEXT NOT NULL,
+  complete INTEGER NOT NULL,
+  next_page INTEGER,
+  skip_reason TEXT,
+  indexed_at INTEGER NOT NULL,
+  error TEXT
+);
+CREATE TABLE chunks (
+  id INTEGER PRIMARY KEY,
+  file_id INTEGER NOT NULL REFERENCES files(id) ON DELETE CASCADE,
+  ordinal INTEGER NOT NULL,
+  page INTEGER,
+  kind TEXT NOT NULL,
+  text TEXT,
+  start_byte INTEGER,
+  end_byte INTEGER
+);
+CREATE VIRTUAL TABLE chunks_fts USING fts5(
+  text, filename, path, tokenize='unicode61'
+);
+CREATE VIRTUAL TABLE chunks_vec USING vec0(
+  embedding FLOAT[{d}],
+  file_kind TEXT,
+  ext TEXT,
+  mtime INTEGER,
+  collection TEXT
+);
+"#
+    )
+}

+    assert_eq!(got.next_page, Some(3));
+
+    // Upsert on the same path updates in place and keeps the id.
+    let mut updated = row;
+    updated.size = 9;
+    updated.complete = false;
+    updated.error = Some("boom".into());
+    let id2 = store.upsert_file(&updated).unwrap();
+    assert_eq!(id2, id);
+    let got2 = store.get_file_by_path("/a/lease.pdf").unwrap().unwrap();
+    assert_eq!(got2.size, 9);
+    assert!(!got2.complete);
+    assert_eq!(got2.error.as_deref(), Some("boom"));
+
+    assert!(store.delete_file_by_path("/a/lease.pdf").unwrap());
+    assert!(store.get_file_by_path("/a/lease.pdf").unwrap().is_none());
+    assert!(!store.delete_file_by_path("/a/lease.pdf").unwrap());
+}

+#[test]
+fn delete_chunk_kind_sides_keep_the_other_side() {
+    let store = open_temp(8);
+    // Text file: only a 'text' chunk. PDF: 'pdf_page' + 'image' chunks.
+    let fid_text = store.upsert_file(&file("/a/t.txt", FileKind::Text)).unwrap();
+    let fid_pdf = store.upsert_file(&file("/a/mix.pdf", FileKind::Pdf)).unwrap();
+    let tx = store.begin().unwrap();
+    let c_text = store.insert_chunk(&tx, chunk(fid_text, 0, ChunkKind::Text, Some("body"))).unwrap();
+    let c_page = store.insert_chunk(&tx, NewChunk {
+        file_id: fid_pdf, ordinal: 0, page: Some(1), kind: ChunkKind::PdfPage,
+        text: Some("page".into()), start_byte: None, end_byte: None,
+    }).unwrap();
+    let c_img = store.insert_chunk(&tx, chunk(fid_pdf, 1, ChunkKind::Image, None)).unwrap();
+    for cid in [c_text, c_page, c_img] {
+        store.insert_vec(&tx, cid, &[0.5; 8], FileKind::Text, "txt", 1, "Documents").unwrap();
+    }
+    tx.commit().unwrap();
+
+    // Text side: delete 'text' chunks, keep 'image' + 'pdf_page'.
+    store.delete_text_chunks_keep_images(fid_pdf).unwrap();
+    let pdf_kinds: Vec<String> = store
+        .conn()
+        .prepare("select kind from chunks where file_id = ? order by id")
+        .unwrap()
+        .query_map([fid_pdf], |r| r.get(0))
+        .unwrap()
+        .collect::<rusqlite::Result<_>>()
+        .unwrap();
+    assert_eq!(pdf_kinds, vec!["pdf_page".to_string(), "image".to_string()]);
+    let text_file_chunks: i64 = store
+        .conn()
+        .query_row("select count(*) from chunks where file_id = ?", [fid_text], |r| r.get(0))
+        .unwrap();
+    assert_eq!(text_file_chunks, 1);
+
+    // Image side: delete 'image' + 'pdf_page' chunks, keep 'text'.
+    store.delete_image_chunks_keep_text(fid_pdf).unwrap();
+    let pdf_left: i64 = store
+        .conn()
+        .query_row("select count(*) from chunks where file_id = ?", [fid_pdf], |r| r.get(0))
+        .unwrap();
+    assert_eq!(pdf_left, 0);
+    store.delete_image_chunks_keep_text(fid_text).unwrap();
+    let text_left: i64 = store
+        .conn()
+        .query_row("select count(*) from chunks where file_id = ?", [fid_text], |r| r.get(0))
+        .unwrap();
+    assert_eq!(text_left, 1);
+    // Vector rows follow their chunks.
+    let vec_rows: i64 = store
+        .conn()
+        .query_row("select count(*) from chunks_vec", [], |r| r.get(0))
+        .unwrap();
+    assert_eq!(vec_rows, 1);
+}

+#[test]
+fn status_counts_reflects_rows() {
+    let store = open_temp(8);
+    let c0 = store.status_counts().unwrap();
+    assert_eq!((c0.files_text, c0.files_image, c0.files_pdf, c0.chunks, c0.incomplete, c0.errors), (0, 0, 0, 0, 0, 0));
+    assert_eq!(c0.indexed_at_max, None);
+
+    let mut t = file("/a/t.txt", FileKind::Text);
+    t.indexed_at = 5;
+    store.upsert_file(&t).unwrap();
+    let mut p = file("/a/p.pdf", FileKind::Pdf);
+    p.complete = false;
+    p.error = Some("boom".into());
+    p.indexed_at = 7;
+    store.upsert_file(&p).unwrap();
+    store.upsert_file(&file("/a/i.png", FileKind::Image)).unwrap();
+    let c1 = store.status_counts().unwrap();
+    assert_eq!((c1.files_text, c1.files_image, c1.files_pdf), (1, 1, 1));
+    assert_eq!(c1.incomplete, 1);
+    assert_eq!(c1.errors, 1);
+    assert_eq!(c1.indexed_at_max, Some(7));
+    assert_eq!(c1.chunks, 0);
+
+    let fid = store.get_file_by_path("/a/t.txt").unwrap().unwrap().id;
+    let tx = store.begin().unwrap();
+    store.insert_chunk(&tx, chunk(fid, 0, ChunkKind::Text, Some("x"))).unwrap();
+    tx.commit().unwrap();
+    let c2 = store.status_counts().unwrap();
+    assert_eq!(c2.chunks, 1);
+}
