+//! Deterministic fake embedder (`FERRET_EMBEDDER=fake`): blake3-derived unit
+//! vectors, independent of any server. Test hook only.
+
+use super::{l2_normalize, EmbedInput, Embedder};
+use crate::error::Result;
+use crate::types::PrefixProfile;
+
+pub struct FakeEmbedder {
+    profile: PrefixProfile,
+    dimensions: u32,
+}
+
+impl FakeEmbedder {
+    /// Test hook with the model-card profile default.
+    pub fn new(dimensions: u32) -> Self {
+        Self::new_with_profile(PrefixProfile::EmbeddingGemma2, dimensions)
+    }
+
+    /// Fake embedder carrying the merged-config profile so vectors follow
+    /// the active prefix strings.
+    pub fn new_with_profile(profile: PrefixProfile, dimensions: u32) -> Self {
+        Self { profile, dimensions }
+    }
+
+    /// `blake3(prefix || 0x1f || payload)`: the prefix is the profile's text
+    /// prompt (empty for image-only inputs), the payload is the UTF-8 text or
+    /// the raw image bytes. A PDF page hashes `image_png || 0x1f || text` so
+    /// it never collides with a plain `Image` of the same bytes.
+    fn hash_input(&self, input: &EmbedInput) -> Vec<u8> {
+        let mut buf = Vec::new();
+        if let Some(prefix) = super::apply_prefix(self.profile, input) {
+            buf.extend_from_slice(prefix.as_bytes());
+        }
+        buf.push(0x1f);
+        match input {
+            EmbedInput::Query { text, .. } | EmbedInput::Document { text, .. } => {
+                buf.extend_from_slice(text.as_bytes());
+            }
+            EmbedInput::Image { bytes, .. } => buf.extend_from_slice(bytes),
+            EmbedInput::PdfPage { image_png, text } => {
+                buf.extend_from_slice(image_png);
+                buf.push(0x1f);
+                if let Some(text) = text {
+                    buf.extend_from_slice(text.as_bytes());
+                }
+            }
+        }
+        buf
+    }
+}
+
+impl Embedder for FakeEmbedder {
+    fn embed(&self, input: EmbedInput) -> Result<Vec<f32>> {
+        let mut hasher = blake3::Hasher::new();
+        hasher.update(&self.hash_input(&input));
+        // First `dimensions` output bytes as `u8 as f32`, always finite; the
+        // extended output covers `dimensions` beyond blake3's 32-byte digest.
+        let mut bytes = vec![0u8; self.dimensions as usize];
+        hasher.finalize_xof().fill(&mut bytes);
+        let raw: Vec<f32> = bytes.iter().map(|&b| b as f32).collect();
+        l2_normalize(raw)
+    }
+
+    fn images_supported(&self) -> bool {
+        true
+    }
+}

+//! Embedder front end: typed inputs, the `Embedder` trait, L2
+//! normalization / dimension enforcement, and `build_embedder` with the
+//! `FERRET_EMBEDDER=fake` test hook. HTTP backends land with Task 12; until
+//! then only the fake embedder can be built.
+
+pub mod fake;
+pub mod prefix;
+

+/// One typed embedding input per modality (spec §3.1). Callers pass a typed
+/// input so prefixes and modalities stay consistent.
+#[derive(Debug, Clone)]
+pub enum EmbedInput {
+    /// Search/code query; `task` selects the query prefix.
+    Query { text: String, task: QueryTask },
+    /// Text document; `title` is the basename, `"none"` in the prefix when
+    /// unknown.
+    Document { title: Option<String>, text: String },
+    /// Image-only input; no text prefix.
+    Image { bytes: Vec<u8>, mime: String },
+    /// One PDF page: image part plus optional page text, one vector.
+    PdfPage { image_png: Vec<u8>, text: Option<String> },
+}
+
+/// An embedding backend. Every output is a single L2-normalized vector of
+/// `embedder.dimensions` finite `f32`s.
+pub trait Embedder: Send + Sync {
+    fn embed(&self, input: EmbedInput) -> Result<Vec<f32>>;
+    fn embed_batch(&self, inputs: &[EmbedInput]) -> Result<Vec<Vec<f32>>> {
+        inputs.iter().cloned().map(|i| self.embed(i)).collect()
+    }
+    fn images_supported(&self) -> bool;
+}
+
+/// Build the embedder for the merged config. `FERRET_EMBEDDER=fake` wins over
+/// `embedder.backend` (any backend) and the fake reports
+/// `images_supported = true`. Without the fake hook this is fatal until the
+/// HTTP backends land (Task 12).
+pub fn build_embedder(
+    cfg: &Config,
+    env: &dyn Fn(&str) -> Option<String>,
+) -> Result<Box<dyn Embedder>> {
+    if env("FERRET_EMBEDDER").as_deref() == Some("fake") {
+        return Ok(Box::new(FakeEmbedder::new_with_profile(
+            cfg.embedder.prefix_profile,
+            cfg.embedder.dimensions,
+        )));
+    }
+    Err(Error::fatal("http embedder not implemented"))
+}
+
+/// L2-normalize `v`. Any non-finite component, or a zero norm, is an error:
+/// ferret stores only finite `f32` and every stored vector is unit length.
+pub fn l2_normalize(v: Vec<f32>) -> Result<Vec<f32>> {
+    if v.iter().any(|x| !x.is_finite()) {
+        return Err(Error::fatal("vector contains non-finite values"));
+    }
+    let norm = v.iter().map(|x| x * x).sum::<f32>().sqrt();
+    if norm == 0.0 {
+        return Err(Error::fatal("cannot L2-normalize a zero vector"));
+    }
+    Ok(v.into_iter().map(|x| x / norm).collect())
+}
+
+/// Enforce the configured `dimensions` on a raw backend vector (spec §3.3).
+/// `openai`/`sglang` receive `dimensions` in the request body and must return
+/// exactly that length. `llama_server`/`litert_lm` do not send `dimensions`:
+/// a longer vector is truncated then L2-renormalized. A shorter vector, or
+/// any non-finite component, is an error.
+pub fn truncate_or_check(v: Vec<f32>, backend: Backend, dimensions: u32) -> Result<Vec<f32>> {
+    if v.iter().any(|x| !x.is_finite()) {
+        return Err(Error::fatal("embedder returned non-finite values"));
+    }
+    let n = dimensions as usize;
+    match backend {
+        Backend::Openai | Backend::Sglang => {
+            if v.len() != n {
+                return Err(Error::fatal(format!(
+                    "embedder returned {} dimensions, expected {dimensions}",
+                    v.len()
+                )));
+            }
+            Ok(v)
+        }
+        Backend::LlamaServer | Backend::LitertLm => {
+            if v.len() < n {
+                return Err(Error::fatal(format!(
+                    "embedder returned {} dimensions, expected at least {dimensions}",
+                    v.len()
+                )));
+            }
+            if v.len() > n {
+                return l2_normalize(v.into_iter().take(n).collect());
+            }
+            Ok(v)
+        }
+    }
+}

+//! Prompt prefixes per profile (spec §3.2). The strings are the model-card /
+//! LiteRT-LM docs spellings verbatim; test vectors pin them, so do not
+//! "improve" them. Prefixes apply to text only: image-only inputs (and a
+//! PDF page with no text) get no text prefix.
+
+use super::EmbedInput;
+use crate::types::{PrefixProfile, QueryTask};
+
+/// Text prompt for `input` under `profile`, or `None` when there is no text
+/// part. A PDF page with text uses the profile's Document prefix (title
+/// `"none"`); a textless page has no text part.
+pub fn apply_prefix(profile: PrefixProfile, input: &EmbedInput) -> Option<String> {
+    match (profile, input) {
+        (
+            PrefixProfile::EmbeddingGemma2,
+            EmbedInput::Query { text, task: QueryTask::Search },
+        ) => Some(format!("task: search result | query: {text}")),
+        (
+            PrefixProfile::EmbeddingGemma2,
+            EmbedInput::Query { text, task: QueryTask::Code },
+        ) => Some(format!("task: code retrieval | query: {text}")),
+        (PrefixProfile::EmbeddingGemma2, EmbedInput::Document { title, text }) => Some(format!(
+            "title: {} | text: {text}",
+            title.as_deref().unwrap_or("none")
+        )),
+        (PrefixProfile::EmbeddingGemma2, EmbedInput::PdfPage { text: Some(text), .. }) => {
+            Some(format!("title: none | text: {text}"))
+        }
+        (
+            PrefixProfile::LitertLm,
+            EmbedInput::Query { text, task: QueryTask::Search },
+        ) => Some(format!("task: search query | text: {text}")),
+        (
+            PrefixProfile::LitertLm,
+            EmbedInput::Query { text, task: QueryTask::Code },
+        ) => Some(format!("task: code retrieval | query: {text}")),
+        (PrefixProfile::LitertLm, EmbedInput::Document { text, .. }) => {
+            Some(format!("task: search result | text: {text}"))
+        }
+        (PrefixProfile::LitertLm, EmbedInput::PdfPage { text: Some(text), .. }) => {
+            Some(format!("task: search result | text: {text}"))
+        }
+        // Image-only inputs, and a PDF page with no text, have no text part.
+        _ => None,
+    }
+}
