+pub fn run_search(

+    let keyword = parsed.keyword_query.trim().to_string();
+    let candidates = cfg.search.candidates.max(1);
+

+    // usage error; hybrid simply skips the vec arm (browse/filter mode).
+    let want_vec = match req.mode {

+    let want_filename = req.mode == SearchMode::Hybrid && !keyword.is_empty();

+    // Built before the vec arm moves `semantic` into the embed input.
+    let terms: Vec<String> = semantic

+        .map(|t| t.to_lowercase())
+        .collect();

+    let mut entries: Vec<(i64, Arm, u32)> = Vec::new();
+    let mut vec_scores: HashMap<i64, f64> = HashMap::new();
+    let mut fts_scores: HashMap<i64, f64> = HashMap::new();
+    let mut fn_scores: HashMap<i64, f64> = HashMap::new();

+        cfg.search.rrf_k,
+        cfg.search.weight_vec,
+        cfg.search.weight_fts,

+    })

+/// ≤200 characters from `text` around the first query-term match window;
+/// with no match, the first 200 characters (spec §6.4).
+fn snippet(text: &str, terms: &[String]) -> String {

+    if chars.len() <= 200 {
+        return text.to_string();
+    }
+    let lower: Vec<char> = text.to_lowercase().chars().collect();
+    let mut best: Option<usize> = None;
+    for term in terms {
+        let needle: Vec<char> = term.chars().collect();

+            continue;
+        }
+        if let Some(pos) = find_subslice(&lower, &needle) {
+            best = Some(best.map_or(pos, |b: usize| b.min(pos)));
+        }

+    let start = best.map_or(0, |p| p.saturating_sub(60));
+    let end = (start + 200).min(chars.len());
+    chars[start..end].iter().collect()

+//! Search query parser.
+//!
+//! Grammar (spec §6): a prefix is a token at the start of the query or after
+//! whitespace — `name:` then a value. `kind:pdf` and `kind: pdf` are the same.
+//! A value runs until the next whitespace, or until the closing quote when
+//! quoted (`path:"/home/nate/My Documents"`). Unknown prefixes stay in the
+//! query text (they are search text, not errors).
+//!
+//! One scan produces three query copies:
+//! - `nl_query` / `keyword_query`: the remainder after prefix stripping,
+//!   verbatim, keeping quotes and FTS operators. The keyword arm builds its
+//!   FTS5 `MATCH` from this copy (later task); JSON `query` prints it.
+//! - `semantic_query`: phrases unquoted (inner text kept) and standalone FTS
+//!   operator tokens dropped — `OR`, `AND`, `NOT`, and tokens an operator
+//!   sign introduces (`+term` / `-term`).
+//!
+//! Hyphens: `+` / `-` are operators only at the START of a token (spec §6.3:
+//! `left-wing` and `COVID-19` are not subtraction). Interior hyphens are
+//! never rewritten.
+
+use std::path::PathBuf;
+
+use chrono::{Local, NaiveDate, NaiveDateTime, NaiveTime, TimeZone};
+
+use crate::config::expand_tilde;
+use crate::error::{Error, Result};
+use crate::types::{FileKind, QueryTask};
+
+/// clap-parsed filter flags. Flags and query prefixes are equivalent
+/// (spec §6.3); flags win on conflict. Repeatable flags union their values.
+#[derive(Debug, Clone, Default, PartialEq)]
+pub struct FilterFlags {
+    pub kind: Vec<FileKind>,
+    pub ext: Vec<String>,
+    pub under: Option<PathBuf>,
+    pub collection: Option<String>,
+    pub name: Option<String>,
+    /// `YYYY-MM-DD`, same semantics as the `after:` prefix.
+    pub after: Option<String>,
+    /// `YYYY-MM-DD`, same semantics as the `before:` prefix.
+    pub before: Option<String>,
+    pub code: bool,
+}
+
+/// SQL-able filters (spec §6.3). All AND-combined by the search query.
+#[derive(Debug, Clone, PartialEq)]
+pub struct Filters {
+    pub kind: Vec<FileKind>,
+    pub ext: Vec<String>, // no dot, lower
+    pub under: Option<PathBuf>,
+    pub collection: Option<String>,
+    pub name: Option<String>, // glob, as given
+    pub after: Option<i64>,   // unix seconds local 00:00:00
+    pub before: Option<i64>,  // next day 00:00:00 local
+}
+
+#[derive(Debug, Clone, PartialEq)]
+pub struct ParsedQuery {
+    pub nl_query: String,       // remainder after prefix strip (JSON `query`)
+    pub semantic_query: String, // unquote phrases, drop standalone OR/AND/NOT
+    pub keyword_query: String,  // remainder including quotes and operators
+    pub task: QueryTask,
+    pub filters: Filters,
+}
+
+/// Parse `raw` into filters plus the semantic/keyword query copies. Flags
+/// override the equivalent prefixes.
+pub fn parse_query(raw: &str, flag_overrides: &FilterFlags) -> Result<ParsedQuery> {
+    let mut kind: Vec<FileKind> = Vec::new();
+    let mut ext: Vec<String> = Vec::new();
+    let mut under: Option<PathBuf> = None;
+    let mut collection: Option<String> = None;
+    let mut name: Option<String> = None;
+    let mut after: Option<i64> = None;
+    let mut before: Option<i64> = None;
+    let mut task = QueryTask::Search;
+
+    // Remainder tokens: (raw form with quotes, Some(inner) when quoted).
+    let mut parts: Vec<(String, Option<String>)> = Vec::new();
+
+    let mut idx = 0;
+    while idx < raw.len() {
+        if raw[idx..].starts_with(char::is_whitespace) {
+            idx += raw[idx..].chars().next().unwrap().len_utf8();
+            continue;
+        }
+
+        // A quoted token is always query text, never a prefix.
+        if raw[idx..].starts_with('"') {
+            let (inner, next) = read_quoted(raw, idx)?;
+            parts.push((format!("\"{inner}\""), Some(inner)));
+            idx = next;
+            continue;
+        }
+
+        // Try a prefix head: identifier chars up to a ':' (no whitespace or
+        // quote may sit between). Anything else is an ordinary word.
+        let (head, has_colon, after_head) = read_prefix_head(raw, idx);
+        if has_colon && is_known_prefix(head) {
+            if head == "code" {
+                // `code:` takes no value: `code: hashmap rust` means
+                // task = code with query "hashmap rust" (spec §6.3).
+                task = QueryTask::Code;
+                idx = after_head;
+                continue;
+            }
+            let (value, value_end) = take_value(raw, after_head, head)?;
+            match head {
+                "kind" | "type" => kind.push(parse_kind(&value)?),
+                "ext" => ext.push(normalize_ext(&value)),
+                "path" | "under" => under = Some(canonical_dir(&value)?),
+                "collection" => collection = Some(value),
+                "name" => name = Some(value),
+                "after" => after = Some(after_midnight(&value)?),
+                "before" => before = Some(before_midnight(&value)?),
+                _ => unreachable!("is_known_prefix filtered the names"),
+            }
+            idx = value_end;
+            continue;
+        }
+
+        let (word, next) = read_word(raw, idx);
+        parts.push((word, None));
+        idx = next;
+    }
+
+    if flag_overrides.code {
+        task = QueryTask::Code;
+    }
+
+    // Flags win on conflict; repeatable flags union their own values.
+    let kind = if flag_overrides.kind.is_empty() {
+        kind
+    } else {
+        flag_overrides.kind.clone()
+    };
+    let ext = if flag_overrides.ext.is_empty() {
+        ext
+    } else {
+        flag_overrides.ext.iter().map(|e| normalize_ext(e)).collect()
+    };
+    let under = match &flag_overrides.under {
+        Some(p) => Some(canonical_dir(&p.to_string_lossy())?),
+        None => under,
+    };
+    let collection = flag_overrides.collection.clone().or(collection);
+    let name = flag_overrides.name.clone().or(name);
+    let after = match &flag_overrides.after {
+        Some(d) => Some(after_midnight(d)?),
+        None => after,
+    };
+    let before = match &flag_overrides.before {
+        Some(d) => Some(before_midnight(d)?),
+        None => before,
+    };
+
+    let joined = || -> String {
+        parts
+            .iter()
+            .map(|(raw, _)| raw.as_str())
+            .collect::<Vec<_>>()
+            .join(" ")
+    };
+
+    Ok(ParsedQuery {
+        nl_query: joined(),
+        semantic_query: semantic_copy(&parts),
+        keyword_query: joined(),
+        task,
+        filters: Filters {
+            kind,
+            ext,
+            under,
+            collection,
+            name,
+            after,
+            before,
+        },
+    })
+}

+/// Known prefix names (spec §6.3 filter table plus `code`).
+fn is_known_prefix(head: &str) -> bool {
+    matches!(
+        head,
+        "kind" | "type" | "ext" | "path" | "under" | "collection" | "name" | "after" | "before" | "code"
+    )
+}

+/// Read the candidate prefix head at `start`: identifier chars up to a ':'.
+/// Returns (head, whether a colon terminated it, index just past the head).
+/// A whitespace or quote terminates the head with no colon.
+fn read_prefix_head(raw: &str, start: usize) -> (&str, bool, usize) {
+    let mut end = start;
+    while end < raw.len() {
+        let ch = raw[end..].chars().next().unwrap();
+        if ch == ':' {
+            return (&raw[start..end], true, end + 1);
+        }
+        if ch.is_whitespace() || ch == '"' {
+            return (&raw[start..end], false, end);
+        }
+        end += ch.len_utf8();
+    }
+    (&raw[start..end], false, end)
+}
+
+/// Read the value for a prefix whose head ended at `at`.
+/// Attached value: `kind:pdf` / quoted `path:"a b"`. Separated value:
+/// `kind: pdf` (whitespace skipped, then quoted or bare word).
+fn take_value(raw: &str, at: usize, pname: &str) -> Result<(String, usize)> {
+    if at < raw.len() && raw[at..].starts_with('"') {
+        return read_quoted(raw, at);
+    }
+    if at < raw.len() && !raw[at..].starts_with(char::is_whitespace) {
+        let (word, end) = read_word(raw, at);
+        return Ok((word, end));
+    }
+    let mut idx = at;
+    while idx < raw.len() && raw[idx..].starts_with(char::is_whitespace) {
+        idx += raw[idx..].chars().next().unwrap().len_utf8();
+    }
+    if idx >= raw.len() {
+        return Err(Error::usage(format!("prefix `{pname}:` requires a value")));
+    }
+    if raw[idx..].starts_with('"') {
+        return read_quoted(raw, idx);
+    }
+    let (word, end) = read_word(raw, idx);
+    Ok((word, end))
+}
+
+/// Read a bare word from `start` to the next whitespace.
+fn read_word(raw: &str, start: usize) -> (String, usize) {
+    let mut end = start;
+    while end < raw.len() && !raw[end..].starts_with(char::is_whitespace) {
+        end += raw[end..].chars().next().unwrap().len_utf8();
+    }
+    (raw[start..end].to_string(), end)
+}
+
+/// Read a "quoted" token from `start` (which must be at the opening quote).
+/// Returns the inner text and the index past the closing quote.
+fn read_quoted(raw: &str, start: usize) -> Result<(String, usize)> {
+    let mut idx = start + 1;
+    let mut inner = String::new();
+    while idx < raw.len() {
+        if raw[idx..].starts_with('"') {
+            return Ok((inner, idx + 1));
+        }
+        let ch = raw[idx..].chars().next().unwrap();
+        inner.push(ch);
+        idx += ch.len_utf8();
+    }
+    Err(Error::usage("unterminated quote in query"))
+}

+fn parse_kind(v: &str) -> Result<FileKind> {
+    v.parse::<FileKind>()
+        .map_err(|e| Error::usage(format!("invalid kind filter: {e}")))
+}

+/// Semantic copy: phrases unquoted, standalone FTS operators dropped.
+///
+/// An operator is a token that is exactly `OR` / `AND` / `NOT`, or that an
+/// operator sign introduces (`+term` / `-term` at token start). `left-wing`
+/// and `COVID-19` never start with an operator sign, so they survive.
+fn semantic_copy(parts: &[(String, Option<String>)]) -> String {
+    let mut tokens: Vec<&str> = Vec::new();
+    for (raw, quoted) in parts {
+        let text = quoted.as_deref().unwrap_or(raw.as_str());
+        for tok in text.split_whitespace() {
+            if is_semantic_operator(tok) {
+                continue;
+            }
+            tokens.push(tok);
+        }
+    }
+    tokens.join(" ")
+}
+
+fn is_semantic_operator(tok: &str) -> bool {
+    matches!(tok, "OR" | "AND" | "NOT") || tok.starts_with('+') || tok.starts_with('-')
+}
+
+fn normalize_ext(e: &str) -> String {
+    e.trim_start_matches('.').to_lowercase()
+}
+
+fn parse_date(s: &str, field: &str) -> Result<NaiveDate> {
+    NaiveDate::parse_from_str(s, "%Y-%m-%d")
+        .map_err(|_| Error::usage(format!("invalid {field} date `{s}` (expected YYYY-MM-DD)")))
+}
+
+fn after_midnight(s: &str) -> Result<i64> {
+    Ok(midnight_ts(parse_date(s, "after")?))
+}
+
+/// `before:` is exclusive-next-day: `mtime <` next day 00:00:00 local, so
+/// `before:2026-01-15` includes all of Jan 14.
+fn before_midnight(s: &str) -> Result<i64> {
+    let d = parse_date(s, "before")?;
+    let next = d
+        .succ_opt()
+        .ok_or_else(|| Error::usage(format!("date `{s}` overflows")))?;
+    Ok(midnight_ts(next))
+}
+
+fn midnight_ts(d: NaiveDate) -> i64 {
+    let dt = NaiveDateTime::new(d, NaiveTime::MIN);
+    Local
+        .from_local_datetime(&dt)
+        .earliest()
+        .map(|t| t.timestamp())
+        .unwrap_or_else(|| dt.and_utc().timestamp())
+}

+        .map_err(|_| Error::usage(format!("path does not exist: {}", expanded.display())))

+//! Reciprocal Rank Fusion (spec §6.2).
+//!
+//! `score(id) = Σ_arm weight_arm / (rrf_k + rank_arm)` with 1-based ranks.
+//! An arm that did not list an id contributes nothing.
+
+/// Which retrieval arm listed a candidate.
+#[derive(Debug, Clone, Copy, PartialEq, Eq)]
+pub enum Arm {
+    Vec,
+    Fts,
+    Filename,
+}
+
+/// Fuse per-arm ranks into RRF scores, best first. Ties break on ascending
+/// id so the output order is deterministic.
+pub fn rrf(
+    entries: &[(i64, Arm, u32)],
+    k: f64,
+    w_vec: f64,
+    w_fts: f64,
+    w_fn: f64,
+) -> Vec<(i64, f64)> {
+    let weight = |arm: Arm| match arm {
+        Arm::Vec => w_vec,
+        Arm::Fts => w_fts,
+        Arm::Filename => w_fn,
+    };
+
+    // First-seen order preserved; sums accumulate per id.
+    let mut fused: Vec<(i64, f64)> = Vec::new();
+    for &(id, arm, rank) in entries {
+        let term = weight(arm) / (k + f64::from(rank));
+        if let Some(slot) = fused.iter_mut().find(|(i, _)| *i == id) {
+            slot.1 += term;
+        } else {
+            fused.push((id, term));
+        }
+    }
+    fused.sort_by(|a, b| b.1.total_cmp(&a.1).then(a.0.cmp(&b.0)));
+    fused
+}
+

+mod tests {

+        // Ranks are 1-based; an id only appears via a real entry.
+        let fused = rrf(&[], 60.0, 1.0, 1.0, 0.4);
+        assert!(fused.is_empty());
+    }
+
+    #[test]
+    fn determinism_by_id_on_ties() {
+        let fused = rrf(&[(9, Arm::Vec, 1), (4, Arm::Vec, 1)], 60.0, 1.0, 1.0, 0.4);
+        assert_eq!(fused[0].0, 4);

+//! Per-arm SQL for search (spec §6.1).
+//!
+//! Filters apply in SQL before ranking, never in Rust after fetch:
+//! `kind` / `ext` / `mtime` / `collection` are vec0 metadata (true KNN
+//! prefilter); `under` / `name` cannot be evaluated during KNN, so when
+//! either is set the vec arm inflates `k`, JOINs `files`, drops mismatches,
+//! and keeps at most `candidates` — never a second KNN loop.
+//!
+//! `ferret_basename(path)` is the scalar function `store::open` registers;
+//! `--name` GLOBs the basename and stays case-sensitive.
+
+use rusqlite::types::Value;
+
+use crate::search::Filters;
+
+/// One candidate row from an arm. `display` is the arm's display score
+/// (vec0 L2 distance as cosine, or raw FTS5 `rank`); ranks come from the
+/// row's position (1-based), never from the score.
+pub struct ArmRow {
+    pub chunk_id: i64,
+    pub display: f64,
+}
+
+/// Escape GLOB metacharacters so a literal string can be embedded in a
+/// pattern (`/a[b` must not start a character class).
+fn glob_escape(s: &str) -> String {
+    let mut out = String::with_capacity(s.len() + 4);
+    for c in s.chars() {
+        match c {
+            '*' | '?' | '[' => {
+                out.push('[');
+                out.push(c);
+                out.push(']');
+            }
+            _ => out.push(c),
+        }
+    }
+    out
+}
+
+/// Little-endian f32 blob for a sqlite-vec `MATCH` argument.
+fn vec_blob(v: &[f32]) -> Vec<u8> {
+    let mut out = Vec::with_capacity(v.len() * 4);
+    for f in v {
+        out.extend_from_slice(&f.to_le_bytes());
+    }
+    out
+}
+
+/// Append the vec0-metadata / files-JOIN filter fragments shared by the
+/// vec arm. `use_files` selects the `f.` column prefix; without it the
+/// columns are the vec0 metadata (`file_kind`, not `kind`).
+fn push_metadata(sql: &mut String, params: &mut Vec<Value>, use_files: bool, filters: &Filters) {
+    let col = |name: &str| {
+        if use_files {
+            format!("f.{name}")
+        } else if name == "kind" {
+            // vec0 metadata column name (files.kind is `kind`).
+            "file_kind".to_string()
+        } else {
+            name.to_string()
+        }
+    };
+    if !filters.kind.is_empty() {
+        let ph = vec!["?"; filters.kind.len()].join(", ");
+        sql.push_str(&format!(" AND {} IN ({ph})", col("kind")));
+        for k in &filters.kind {
+            params.push(Value::Text(k.to_string()));
+        }
+    }
+    if !filters.ext.is_empty() {
+        let ph = vec!["?"; filters.ext.len()].join(", ");
+        sql.push_str(&format!(" AND {} IN ({ph})", col("ext")));
+        for e in &filters.ext {
+            params.push(Value::Text(e.clone()));
+        }
+    }
+    if let Some(after) = filters.after {
+        sql.push_str(&format!(" AND {} >= ?", col("mtime")));
+        params.push(Value::Integer(after));
+    }
+    if let Some(before) = filters.before {
+        sql.push_str(&format!(" AND {} < ?", col("mtime")));
+        params.push(Value::Integer(before));
+    }
+    if let Some(collection) = &filters.collection {
+        sql.push_str(&format!(" AND {} = ?", col("collection")));
+        params.push(Value::Text(collection.clone()));
+    }
+    if use_files {
+        if let Some(under) = &filters.under {
+            // Canonical prefix; trailing '/' trimmed so `path = u` and
+            // `path GLOB u||'/*'` cover the root itself and its contents.
+            let u = under.to_string_lossy().trim_end_matches('/').to_string();
+            sql.push_str(" AND (f.path = ? OR f.path GLOB ? || '/*')");
+            params.push(Value::Text(glob_escape(&u)));
+            params.push(Value::Text(glob_escape(&u)));
+        }
+        if let Some(name) = &filters.name {
+            // Basename only; the user's GLOB is used as given (case-sensitive).
+            sql.push_str(" AND ferret_basename(f.path) GLOB ?");
+            params.push(Value::Text(name.clone()));
+        }
+    }
+}

+/// Vec arm: KNN over `chunks_vec` with metadata prefilter; JOIN `files` and
+/// inflate `k` only when `--under` / `--name` is set.
+pub fn vec_arm(
+    conn: &rusqlite::Connection,
+    embedding: &[f32],
+    filters: &Filters,
+    candidates: usize,
+) -> Result<Vec<ArmRow>, rusqlite::Error> {
+    let joined = filters.under.is_some() || filters.name.is_some();
+    let k: i64 = if joined {
+        (candidates as i64 * 8).min(400)
+    } else {
+        candidates as i64
+    };
+
+    let mut sql = String::from("SELECT cv.rowid, cv.distance FROM chunks_vec cv");
+    let mut params: Vec<Value> = vec![Value::Blob(vec_blob(embedding)), Value::Integer(k)];
+    if joined {
+        sql.push_str(" JOIN chunks c ON c.id = cv.rowid JOIN files f ON f.id = c.file_id");
+    }
+    sql.push_str(" WHERE cv.embedding MATCH ? AND cv.k = ?");
+    push_metadata(&mut sql, &mut params, joined, filters);
+    sql.push_str(" ORDER BY cv.distance");
+    if joined {
+        sql.push_str(" LIMIT ?");
+        params.push(Value::Integer(candidates as i64));
+    }
+
+    let mut stmt = conn.prepare(&sql)?;
+    let rows = stmt.query_map(rusqlite::params_from_iter(params.iter()), |r| {
+        let d: f64 = r.get(1)?;
+        // Vectors are L2-normalized, so cosine = 1 - d²/2, clamped.
+        let cosine = (1.0 - (d * d) / 2.0).clamp(0.0, 1.0);
+        Ok(ArmRow {
+            chunk_id: r.get(0)?,
+            display: cosine,
+        })
+    })?;
+    rows.collect()
+}
+
+/// Keyword arm: FTS5 MATCH. `column_filter` is an FTS5 column selector like
+/// `{filename path}` for the filename arm (matches only those columns),
+/// or `None` for the plain body match. Candidates are capped at
+/// `candidates` after ordering by FTS5 `rank`.
+pub fn fts_arm(
+    conn: &rusqlite::Connection,
+    query: &str,
+    filters: &Filters,
+    candidates: usize,
+    column_filter: Option<&str>,
+) -> Result<Vec<ArmRow>, rusqlite::Error> {
+    let match_arg = match column_filter {
+        // FTS5 column filters require the colon: `{a b}: query`.
+        Some(cols) => format!("{cols}: {query}"),
+        None => query.to_string(),
+    };
+    let mut sql = String::from(
+        "SELECT fts.rowid, fts.rank FROM chunks_fts fts \
+         JOIN chunks c ON c.id = fts.rowid JOIN files f ON f.id = c.file_id \
+         WHERE chunks_fts MATCH ?",
+    );
+    let mut params: Vec<Value> = vec![Value::Text(match_arg)];
+    push_metadata(&mut sql, &mut params, true, filters);
+    sql.push_str(" ORDER BY fts.rank LIMIT ?");
+    params.push(Value::Integer(candidates as i64));
+
+    let mut stmt = conn.prepare(&sql)?;
+    let rows = stmt.query_map(rusqlite::params_from_iter(params.iter()), |r| {
+        Ok(ArmRow {
+            chunk_id: r.get(0)?,
+            display: r.get(1)?,
+        })
+    })?;
+    rows.collect()
+}

+//! Task 8: RRF fusion.
+
+use ferret::search::rrf::{rrf, Arm};
+
+#[test]
+fn rrf_prefers_agreement() {
+    // chunk 1 rank 1 vec + rank 1 fts; chunk 2 rank 2 vec only
+    let fused = rrf(
+        &[(1, Arm::Vec, 1), (1, Arm::Fts, 1), (2, Arm::Vec, 2)],
+        60.0, 1.0, 1.0, 0.4,
+    );
+    assert_eq!(fused[0].0, 1);
+    let s1 = 1.0 / 61.0 + 1.0 / 61.0;
+    assert!((fused[0].1 - s1).abs() < 1e-12);
+}
+
+#[test]
+fn rrf_orders_by_fused_score_desc() {
+    let fused = rrf(
+        &[(7, Arm::Vec, 1), (9, Arm::Fts, 1), (7, Arm::Fts, 5)],
+        60.0, 1.0, 1.0, 0.4,
+    );
+    // 7: 1/61 + 1/65; 9: 1/61 -> 7 wins on the extra arm
+    assert_eq!(fused[0].0, 7);
+    assert_eq!(fused[1].0, 9);
+    assert!(fused[0].1 > fused[1].1);
+}
+
+#[test]
+fn rrf_filename_weight_applies() {
+    let fused = rrf(&[(1, Arm::Filename, 1)], 60.0, 1.0, 1.0, 0.4);
+    assert!((fused[0].1 - 0.4 / 61.0).abs() < 1e-12);
+}

+#[test]
+fn rrf_same_id_multiple_entries_sum() {
+    let fused = rrf(
+        &[(3, Arm::Vec, 2), (3, Arm::Fts, 3)],
+        60.0, 1.0, 1.0, 0.4,
+    );
+    let expect = 1.0 / 62.0 + 1.0 / 63.0;
